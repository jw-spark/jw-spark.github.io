// ---------- Fetch historical rows from Supabase ---------- // Fetches multiple pages because Supabase/PostgREST may cap each response, // commonly at 1,000 rows even when a larger limit is requested. async function fetchHistoryRows(stop) { const predictionConfig = CONFIG.PREDICTION || {}; // Total historical records to retrieve for this bus stop. const maxRows = predictionConfig.QUERY_LIMIT || 3000; // Keep this at or below the Supabase API maximum rows per response. const pageSize = predictionConfig.PAGE_SIZE || 1000; const allRows = []; let offset = 0; let pageNumber = 1; // Do not retrieve records whose collection time is in the future. const nowIso = new Date().toISOString(); while (allRows.length < maxRows) { const remaining = maxRows - allRows.length; const currentPageSize = Math.min(pageSize, remaining); const url = CONFIG.SUPABASE_URL + "/rest/v1/bus_arrivals" + "?select=bus_stop_code,service_no,sequence,estimated_arrival,collected_at" + "&bus_stop_code=eq." + encodeURIComponent(stop) + "&sequence=eq.next" + "&collected_at=lte." + encodeURIComponent(nowIso) + "&order=collected_at.desc" + "&limit=" + currentPageSize + "&offset=" + offset; log( "Fetching history page " + pageNumber + ": offset=" + offset + ", limit=" + currentPageSize ); const resp = await fetch(url, { headers: { "apikey": CONFIG.SUPABASE_KEY, "Authorization": "Bearer " + CONFIG.SUPABASE_KEY, "Accept": "application/json" } }); if (!resp.ok) { const t = await resp.text(); throw new Error( "Supabase history HTTP " + resp.status + " -> " + t ); } const pageRows = await resp.json(); log( "Page " + pageNumber + " returned " + pageRows.length + " rows" ); if (!Array.isArray(pageRows) || pageRows.length === 0) { log("No more historical rows available."); break; } allRows.push(...pageRows); // A short page means that the server has no more records. if (pageRows.length < currentPageSize) { log("Reached the oldest available record."); break; } offset += pageRows.length; pageNumber++; } log("Total Supabase history rows loaded: " + allRows.length); return allRows.slice(0, maxRows); }
